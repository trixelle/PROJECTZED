<script>
  (function () {
    var TICK = [
      'INBOUND \u2014 analyst, Bay 8: "His last three deferrals came in under a minute"',
      'INBOUND \u2014 Command Line: Relay map triaged.',
      'INBOUND \u2014 CEDD: Stability review window revised. 2.3% improvement.',
      'INBOUND \u2014 Gangplank: What is JH-4N? File that report tonight.',
      'INBOUND \u2014 ESBEN-LR: "Have you...met Ashe?"',
      'INBOUND \u2014 no origin: [Zed is asleep. leave it until morning.]'
    ];
    var box = document.getElementById('tx-tick'), i = 0;
    function nextTick() {
      var s = document.createElement('span');
      s.textContent = TICK[i % TICK.length];
      if (TICK[i % TICK.length].indexOf('no origin') > -1) s.style.color = 'var(--tx-trixe)';
      i++;
      box.appendChild(s);
      box.animate([{ transform: 'translateY(0)' }, { transform: 'translateY(-22px)' }],
        { duration: 520, easing: 'cubic-bezier(.6,0,.2,1)', fill: 'forwards' })
        .onfinish = function () {
          if (box.firstChild) box.removeChild(box.firstChild);
          box.style.transform = 'none'; box.getAnimations().forEach(function (a) { a.cancel(); });
        };
    }
    nextTick(); setInterval(nextTick, 7400);

    var lat = document.getElementById('tx-cl');
    setInterval(function () { lat.textContent = (2 + Math.floor(Math.random() * 7)) + 'ms'; }, 1900);
  })();
</script>
