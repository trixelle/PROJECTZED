<script>
(function(){
function txExtras(){
  var dot=document.getElementById('tx-dot'), btn=document.getElementById('tx-audio');
  if(!dot||!btn) return;
  try{
    if(dot.parentNode!==document.body) document.body.appendChild(dot);
    if(btn.parentNode!==document.body) document.body.appendChild(btn);
  }catch(e){}

  /* ---------------- cursor follower ---------------- */
  var tx=innerWidth/2, ty=innerHeight/2, dx=tx, dy=ty, near=false;
  addEventListener('pointermove',function(e){
    tx=e.clientX; ty=e.clientY;
    var el=e.target, n=!!(el&&el.closest&&el.closest('a,button,summary,.rd,#tx-nav a,.op summary,.tx-tabs button,#tx-gauge .node'));
    if(n!==near){ near=n; dot.classList.toggle('near',n); if(n) TXA.tick(); }
  },{passive:true});

  function follow(){
    var ax=tx-dx, ay=ty-dy;
    if(Math.abs(ax)<0.1 && Math.abs(ay)<0.1) return;   /* idle: no writes */
    dx+=ax*0.16; dy+=ay*0.16;
    dot.style.transform='translate3d('+dx+'px,'+dy+'px,0)';
  }
  if(window.TX) window.TX.add(follow);
  else (function r(){ follow(); requestAnimationFrame(r); })();

  /* generated audio */
  var AC=window.AudioContext||window.webkitAudioContext, ctx=null, master=null, amb=null, on=false, built=false;
  var DEFAULT_ON = false;      
  var AMBIENT_LEVEL = 0.30;   
  function build(){
    if(built) return; built=true;
    ctx=new AC();
    master=ctx.createGain(); master.gain.value=0; master.connect(ctx.destination);
    amb=ctx.createGain(); amb.gain.value=AMBIENT_LEVEL; amb.connect(master);

    var lp=ctx.createBiquadFilter(); lp.type='lowpass'; lp.frequency.value=210; lp.Q.value=6;
    lp.connect(amb);
    [55,82.5,110].forEach(function(f,i){
      var o=ctx.createOscillator(), g=ctx.createGain();
      o.type='sine'; o.frequency.value=f+(i?0.3:0);
      g.gain.value=[0.5,0.28,0.14][i];
      o.connect(g); g.connect(lp); o.start();
    });
    var lfo=ctx.createOscillator(), la=ctx.createGain();
    lfo.frequency.value=0.045; la.gain.value=60;
    lfo.connect(la); la.connect(lp.frequency); lfo.start();


    var n=ctx.sampleRate*2, buf=ctx.createBuffer(1,n,ctx.sampleRate), d=buf.getChannelData(0);
    for(var i=0;i<n;i++) d[i]=(Math.random()*2-1)*0.5;
    var src=ctx.createBufferSource(); src.buffer=buf; src.loop=true;
    var bp=ctx.createBiquadFilter(); bp.type='bandpass'; bp.frequency.value=1400; bp.Q.value=0.6;
    var ng=ctx.createGain(); ng.gain.value=0.06;
    src.connect(bp); bp.connect(ng); ng.connect(amb); src.start();
  }

  function blip(freq,dur,vol,type){
    if(!on||!ctx) return;
    var o=ctx.createOscillator(), g=ctx.createGain();
    o.type=type||'sine'; o.frequency.value=freq;
    g.gain.setValueAtTime(0,ctx.currentTime);
    g.gain.linearRampToValueAtTime(vol,ctx.currentTime+0.012);
    g.gain.exponentialRampToValueAtTime(0.0001,ctx.currentTime+dur);
    o.connect(g); g.connect(master); o.start(); o.stop(ctx.currentTime+dur+0.02);
  }

  var lastTick=0;
  var TXA = window.TXA = {
    start:function(){ if(DEFAULT_ON) TXA.set(true); },
    set:function(v){
      if(v){ build(); if(ctx.state==='suspended') ctx.resume();
             master.gain.setTargetAtTime(0.5,ctx.currentTime,1.2); }
      else if(ctx){ master.gain.setTargetAtTime(0,ctx.currentTime,0.4); }
      on=v;
      btn.classList.toggle('on',v);
      btn.setAttribute('aria-pressed',v?'true':'false');
      document.getElementById('tx-audio-l').textContent = v?'AUDIO ON':'AUDIO OFF';
      try{ localStorage.setItem('tx-audio', v?'1':'0'); }catch(e){}
    },
    tick:function(){
      var t=Date.now(); if(t-lastTick<110) return; lastTick=t;
      blip(1180,0.05,0.014,'square');
    },
    chime:function(){ blip(659.25,0.5,0.05); setTimeout(function(){ blip(987.77,0.4,0.03); },90); }
  };

  btn.addEventListener('click',function(){ TXA.set(!on); });

  try{ if(localStorage.getItem('tx-audio')==='0') DEFAULT_ON=false; }catch(e){}

  if(!document.getElementById('tx-boot')) TXA.start();
  else{
    var mo=new MutationObserver(function(){
      if(!document.getElementById('tx-boot')){ mo.disconnect(); TXA.start(); }
    });
    mo.observe(document.body,{childList:true});
  }
}

if(document.readyState==='loading') document.addEventListener('DOMContentLoaded',txExtras);
else txExtras();
})();
</script>
