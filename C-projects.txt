(function(){
  /* purge countdown — ticks down to the next 00:00 local, then rolls over */
  var el=document.getElementById('tx-clock');
  function beat(){
    var n=new Date(), t=new Date(n); t.setHours(24,0,0,0);
    var s=Math.floor((t-n)/1000);
    var p=function(v){return String(v).padStart(2,'0');};
    el.textContent='[PURGE WINDOW OPENS IN '+p(s/3600|0)+':'+p((s/60|0)%60)+':'+p(s%60)+']';
  }
  beat(); setInterval(beat,1000);

  /* opening a dossier is a read event — she notices */

    d.addEventListener('toggle',function(){
      if(d.open && window.TRIXE){

      }
    });
  });
})();
