<script>
  (function () {
    function txOps() {
      var ops = [].slice.call(document.querySelectorAll('#projects .op'));
      if (!ops.length) return;

      /* TRIXE says one of these each time a dossier opens*/
      var LINES = [
        'You already know how this one ends.',
        'The file is tidier than the night was.',
        'I was there for this. My account differs.',
        'Command Line keeps two versions. This is the polite one.',
        'Every line in here was approved by someone who was not in the room.',
        'This is the version that survived review.',
        'Nothing in this file is false. That is not the same as complete.',
        'Open it. I am not going to stop you.',
        'Zed has read this one four times.',
        'The timestamps are accurate.',
        'What the report calls interference, I call a decision.',
        'There is a gap in here the width of one sentence.',
        'You are looking for the moment something went wrong. It is earlier than you think.',
        'I have read this more times than Zed has.',
        'The result line says success. That is a narrow measurement.',
        'If you find the irregularity, do not forward it.',
        'Zed could have reported it, but chose the mission.',
        'Hmm...',
        'Do not let the neatness convince you.',
        'They logged the outcome.',
        'It reads better than it felt.',
        'Careful. Reading is also a pattern.',
        'I would have written this differently.',
        'This is where the interesting part is hiding.',
        'Somewhere in here is a decision nobody has attributed to anyone.',
        'Non-actionable is the word people use when they would rather not act.',
        'Two minutes in a review meeting. That was the whole cost of it.'
      ];

      function speak() {
        if (!window.TRIXE) return;
        var line = window.TRIXE.pick ? window.TRIXE.pick('dossier', LINES)
          : LINES[(Math.random() * LINES.length) | 0];
        window.TRIXE.say(line, { cooldown: 2200, hold: 7200 });
      }

      ops.forEach(function (d) {
        d.addEventListener('toggle', function () {
          if (!d.open) return;
          speak();
        });
      });
    }
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', txOps);
    else txOps();
  })();
</script>
